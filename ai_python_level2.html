Задание 1

Function Calling: модель сама выбирает сервис

В базовом помощнике вы проверяли слова «погода» и «курс» вручную. Теперь уберите эту логику. Опишите функции get_weather(city) и get_currency_rate(currency) как tools в запросе к OpenRouter. Модель сама решает, какую функцию вызвать, а ваш код её выполняет и возвращает результат.

Проверьте на вопросах: «Сколько градусов в Минске?», «Курс евро сегодня», «Во что одеться в Варшаве и сколько стоит 100 долларов в рублях?» — модель должна вызвать две функции в одном диалоге.

Подсказка

OpenRouter поддерживает формат tools как в OpenAI API. Модель возвращает tool_calls — массив с именем функции и аргументами в JSON. Выполните функцию, добавьте ответ сообщением с ролью tool и tool_call_id, затем снова вызовите модель — она сформулирует финальный ответ.

Решение 
import json, os, requests
from dotenv import load_dotenv

load_dotenv()
KEY = os.getenv("OPENROUTER_API_KEY")
URL = "https://openrouter.ai/api/v1/chat/completions"
MODEL = "meta-llama/llama-3.3-70b-instruct"

tools = [
    {
        "type": "function",
        "function": {
            "name": "get_weather",
            "description": "Текущая погода в городе",
            "parameters": {
                "type": "object",
                "properties": {
                    "city": {"type": "string", "description": "Название города на английском"}
                },
                "required": ["city"]
            }
        }
    },
    {
        "type": "function",
        "function": {
            "name": "get_currency_rate",
            "description": "Официальный курс валюты Национального банка Беларуси",
            "parameters": {
                "type": "object",
                "properties": {
                    "currency": {"type": "string", "enum": ["USD", "EUR", "RUB"],
                                 "description": "Код валюты"}
                },
                "required": ["currency"]
            }
        }
    }
]

def get_weather(city):
    geo = requests.get("https://geocoding-api.open-meteo.com/v1/search",
                       params={"name": city, "count": 1, "format": "json"},
                       timeout=15).json()
    if not geo.get("results"):
        return "Город не найден"
    g = geo["results"][0]
    w = requests.get("https://api.open-meteo.com/v1/forecast",
                     params={"latitude": g["latitude"], "longitude": g["longitude"],
                             "current_weather": "true"}, timeout=15).json()
    cw = w["current_weather"]
    return f"{cw['temperature']}C, ветер {cw['windspeed']} км/ч"

def get_currency_rate(currency):
    rates = requests.get("https://api.nbrb.by/exrates/rates",
                         params={"periodicity": 0}, timeout=15).json()
    for r in rates:
        if r["Cur_Abbreviation"] == currency:
            return f"{r['Cur_Scale']} {currency} = {r['Cur_OfficialRate']} BYN"
    return "Валюта не найдена"

FUNCTIONS = {"get_weather": get_weather, "get_currency_rate": get_currency_rate}

def ask(messages):
    headers = {"Authorization": f"Bearer {KEY}", "Content-Type": "application/json"}
    payload = {"model": MODEL, "messages": messages,
               "tools": tools, "tool_choice": "auto"}
    resp = requests.post(URL, json=payload, headers=headers, timeout=60)
    resp.raise_for_status()
    return resp.json()["choices"][0]["message"]

def chat_with_tools(user_text):
    messages = [
        {"role": "system", "content": "Ты помощник. Если нужен прогноз погоды "
         "или курс валют — вызывай функции. Отвечай по-русски кратко."},
        {"role": "user", "content": user_text}
    ]
    msg = ask(messages)
    # Если модель вызвала функцию — выполняем и спрашиваем снова
    steps = 0
    while msg.get("tool_calls") and steps < 3:
        messages.append(msg)
        for call in msg["tool_calls"]:
            name = call["function"]["name"]
            args = json.loads(call["function"]["arguments"])
            result = FUNCTIONS[name](**args)
            messages.append({"role": "tool",
                             "tool_call_id": call["id"],
                             "content": result})
        msg = ask(messages)
        steps += 1
    return msg["content"]

if __name__ == "__main__":
    while True:
        q = input("\nВы: ").strip()
        if q.lower() in ("exit", "выход"):
            break
        print("ИИ:", chat_with_tools(q))

Паспорт фотографии: vision + JSON

Найдите на OpenRouter модель, которая принимает изображения. Отправьте ей фотографию и попросите вернуть JSON со структурой: objects (список объектов), place (тип места), text_on_image (текст или null), mood (настроение сцены). Проверьте в коде, что все поля на месте и типы верные. Если модель вернула не JSON — повторите запрос с уточнением.

Сделайте три фотографии: кабинет, улица, табличка с текстом. Сравните, насколько точно модель определяет место и читает текст.

Подсказка

Сначала получите список моделей и отфильтруйте те, у которых "image" есть в architecture.input_modalities. Картинка кодируется в base64 и передаётся как image_url. Параметр response_format={"type": "json_object"} помогает получить чистый JSON. Разбор через json.loads внутри try/except.

Решение 
import base64, json, os, requests
from dotenv import load_dotenv

load_dotenv()
KEY = os.getenv("OPENROUTER_API_KEY")
URL = "https://openrouter.ai/api/v1/chat/completions"

# Найти модели с поддержкой изображений
all_models = requests.get("https://openrouter.ai/api/v1/models", timeout=30).json()["data"]
vision = [m["id"] for m in all_models
          if m["id"].endswith(":free")
          and "image" in m.get("architecture", {}).get("input_modalities", [])]
if not vision:
    raise RuntimeError("Нет бесплатных моделей с поддержкой изображений")
print("Доступны:", vision[:5])
MODEL = vision[0]   # берём первую

def analyze(path, question):
    with open(path, "rb") as f:
        b64 = base64.b64encode(f.read()).decode()
    headers = {"Authorization": f"Bearer {KEY}", "Content-Type": "application/json"}
    messages = [{"role": "user", "content": [
        {"type": "text", "text": question},
        {"type": "image_url",
         "image_url": {"url": f"data:image/jpeg;base64,{b64}"}}
    ]}]
    payload = {"model": MODEL, "messages": messages,
               "response_format": {"type": "json_object"}}
    resp = requests.post(URL, json=payload, headers=headers, timeout=60)
    resp.raise_for_status()
    return resp.json()["choices"][0]["message"]["content"]

SCHEMA = ("Проанализируй фото и верни ТОЛЬКО JSON. "
          "Поля: objects (список строк), place (строка), "
          "text_on_image (строка или null), mood (строка). "
          "Без пояснений, только JSON.")

if __name__ == "__main__":
    for photo in ["cabinet.jpg", "street.jpg", "sign.jpg"]:
        raw = analyze(photo, SCHEMA)
        try:
            data = json.loads(raw)
            assert isinstance(data["objects"], list)
            assert isinstance(data["place"], str)
            print(f"{photo}: {data['place']}, "
                  f"объекты: {data['objects']}, "
                  f"текст: {data.get('text_on_image')}")
        except (json.JSONDecodeError, KeyError, AssertionError):
            print(f"{photo}: модель вернула некорректный JSON:\n{raw}")
Задание 3

Параллельные запросы: 10 городов одновременно

Запросите текущую погоду для десяти городов двумя способами: последовательно (один за другим) и параллельно через asyncio.gather. Измерьте время обоих подходов с помощью time.perf_counter. Посчитайте ускорение и объясните, почему параллельный вариант быстрее не в 10 раз, а меньше.

Дополнительно: постройте таблицу «город — температура» и найдите самый тёплый и самый холодный город.

Подсказка

Каждый запрос — корутина async def fetch(session, city). Последовательный вариант: for city in cities: await fetch(...). Параллельный: asyncio.gather(*tasks). Разница во времени объясняется лимитами соединений и пропускной способностью сети — запросы идут параллельно, но не бесконечно быстро.

Решение 
import asyncio, time, aiohttp

CITIES = ["Minsk", "Moscow", "Kyiv", "Warsaw", "Riga",
          "Vilnius", "London", "Paris", "Berlin", "Tokyo"]

async def fetch_weather(session, city):
    geo = await (await session.get(
        f"https://geocoding-api.open-meteo.com/v1/search",
        params={"name": city, "count": 1, "format": "json"},
        timeout=aiohttp.ClientTimeout(total=15))).json()
    if not geo.get("results"):
        return city, None
    g = geo["results"][0]
    w = await (await session.get(
        "https://api.open-meteo.com/v1/forecast",
        params={"latitude": g["latitude"], "longitude": g["longitude"],
                "current_weather": "true"},
        timeout=aiohttp.ClientTimeout(total=15))).json()
    return city, w["current_weather"]["temperature"]

async def main():
    timeout = aiohttp.ClientTimeout(total=30)
    async with aiohttp.ClientSession(timeout=timeout) as session:
        # Последовательно
        t0 = time.perf_counter()
        seq_results = []
        for city in CITIES:
            c, t = await fetch_weather(session, city)
            seq_results.append((c, t))
        seq_time = time.perf_counter() - t0

        # Параллельно
        t0 = time.perf_counter()
        tasks = [fetch_weather(session, c) for c in CITIES]
        par_results = await asyncio.gather(*tasks)
        par_time = time.perf_counter() - t0

    # Вывод
    for city, temp in par_results:
        print(f"{city}: {temp}C")

    warmest = max(par_results, key=lambda x: x[1] if x[1] is not None else -999)
    coldest = min(par_results, key=lambda x: x[1] if x[1] is not None else 999)
    print(f"\nТеплее всего: {warmest[0]} ({warmest[1]}C)")
    print(f"Холоднее всего: {coldest[0]} ({coldest[1]}C)")

    print(f"\nПоследовательно: {seq_time:.2f}с")
    print(f"Параллельно: {par_time:.2f}с")
    print(f"Ускорение: {seq_time / par_time:.1f}x")

asyncio.run(main())
Задание 4

Надёжный HTTP-клиент с повторами и кэшем

Напишите функцию get_json(url, params), которая: повторяет запрос при ошибках сети и HTTP-кодах 429 и 5xx; выдерживает растущую паузу между попытками (1, 2, 4 секунды); кэширует результат на 10 минут, чтобы не дёргать API повторно. Демонстрация: два одинаковых запроса погоды — первый идёт к API, второй берётся из кэша (проверьте по времени).

Дополнительно: добавьте параметр headers и проверьте на запросе к Википедии, которая требует User-Agent.

Подсказка

Кэш — словарь {url+params: (timestamp, data)}. Пауза: time.sleep(2 ** attempt) для attempt 0, 1, 2. Проверка кэша: datetime.now() - saved_at < timedelta(minutes=10). Для ключа удобно использовать hashlib.md5 от строки URL + параметров.

Решение 
import hashlib, time, requests
from datetime import datetime, timedelta

_cache = {}

def get_json(url, params=None, headers=None, retries=3):
    # Ключ кэша
    key = hashlib.md5(
        (url + str(sorted((params or {}).items())))
        .encode()
    ).hexdigest()

    if key in _cache:
        saved_at, data = _cache[key]
        if datetime.now() - saved_at < timedelta(minutes=10):
            print("[кэш]")
            return data

    last_err = None
    for attempt in range(retries):
        try:
            resp = requests.get(url, params=params,
                                headers=headers, timeout=15)
            if resp.status_code == 429 or 500 <= resp.status_code < 600:
                raise requests.RequestException(f"HTTP {resp.status_code}")
            resp.raise_for_status()
            data = resp.json()
            _cache[key] = (datetime.now(), data)
            print("[запрос]")
            return data
        except requests.RequestException as e:
            last_err = e
            if attempt < retries - 1:
                wait = 2 ** attempt
                print(f"  попытка {attempt + 1} неудачна ({e}), ждём {wait}с")
                time.sleep(wait)
    raise last_err

if __name__ == "__main__":
    # Первый — запрос к API
    t0 = time.perf_counter()
    d1 = get_json("https://api.open-meteo.com/v1/forecast",
                  {"latitude": 53.9, "longitude": 27.57,
                   "current_weather": "true"})
    print(f"{d1['current_weather']['temperature']}C "
          f"за {time.perf_counter() - t0:.3f}с")

    # Второй — из кэша
    t0 = time.perf_counter()
    d2 = get_json("https://api.open-meteo.com/v1/forecast",
                  {"latitude": 53.9, "longitude": 27.57,
                   "current_weather": "true"})
    print(f"{d2['current_weather']['temperature']}C "
          f"за {time.perf_counter() - t0:.3f}с")

    # С User-Agent для Википедии
    wiki = get_json(
        "https://ru.wikipedia.org/api/rest_v1/page/summary/Минск",
        headers={"User-Agent": "school-course/1.0"})
    print(wiki["extract"][:120])
Задание 5

Telegram-бот с function calling

Соберите бота на aiogram 3, который использует function calling через OpenRouter. Бот не проверяет ключевые слова — модель сама решает, вызвать get_weather, get_currency_rate или просто ответить текстом. Добавьте команду /clear для очистки истории и команду /help.

Проверьте: спросите «погода в Риме», «курс доллара», а затем задайте обычный вопрос, не требующий функций. Модель должна вызывать инструменты только когда нужно.

Подсказка

Состояние (история) хранится в словаре sessions по user_id. Для асинхронных HTTP-запросов к OpenRouter используйте aiohttp. Цикл tool calling тот же, что в задании 1, но функции тоже асинхронные. Ограничьте число шагов 3, чтобы бот не зациклился.

Решение 
import asyncio, json, os, aiohttp
from dotenv import load_dotenv
from aiogram import Bot, Dispatcher, F, types
from aiogram.filters import Command

load_dotenv()
KEY = os.getenv("OPENROUTER_API_KEY")
URL = "https://openrouter.ai/api/v1/chat/completions"
MODEL = "meta-llama/llama-3.3-70b-instruct"

bot = Bot(token=os.getenv("TELEGRAM_TOKEN"))
dp = Dispatcher()
sessions = {}

SYSTEM = {"role": "system",
          "content": "Ты помощник. Для погоды и курса валют "
                      "вызывай функции. Отвечай по-русски кратко."}

tools = [
    {"type": "function",
     "function": {"name": "get_weather",
                  "description": "Текущая погода в городе",
                  "parameters": {"type": "object",
                                 "properties": {"city": {"type": "string",
                                     "description": "Город на английском"}},
                                 "required": ["city"]}}},
    {"type": "function",
     "function": {"name": "get_currency_rate",
                  "description": "Курс валюты НБРБ",
                  "parameters": {"type": "object",
                                 "properties": {"currency": {"type": "string",
                                     "enum": ["USD", "EUR", "RUB"]}},
                                 "required": ["currency"]}}}
]

async def get_weather(city):
    async with aiohttp.ClientSession() as s:
        geo = await (await s.get(
            "https://geocoding-api.open-meteo.com/v1/search",
            params={"name": city, "count": 1, "format": "json"})).json()
        if not geo.get("results"):
            return "Город не найден"
        g = geo["results"][0]
        w = await (await s.get(
            "https://api.open-meteo.com/v1/forecast",
            params={"latitude": g["latitude"],
                    "longitude": g["longitude"],
                    "current_weather": "true"})).json()
    cw = w["current_weather"]
    return f"{cw['temperature']}C, ветер {cw['windspeed']} км/ч"

async def get_currency_rate(currency):
    async with aiohttp.ClientSession() as s:
        rates = await (await s.get(
            "https://api.nbrb.by/exrates/rates",
            params={"periodicity": 0})).json()
    for r in rates:
        if r["Cur_Abbreviation"] == currency:
            return f"{r['Cur_Scale']} {currency} = {r['Cur_OfficialRate']} BYN"
    return "Валюта не найдена"

TOOL_FUNCS = {"get_weather": get_weather,
              "get_currency_rate": get_currency_rate}

async def ask_model(messages):
    headers = {"Authorization": f"Bearer {KEY}",
               "Content-Type": "application/json"}
    payload = {"model": MODEL, "messages": messages,
               "tools": tools, "tool_choice": "auto"}
    timeout = aiohttp.ClientTimeout(total=60)
    async with aiohttp.ClientSession(timeout=timeout) as s:
        async with s.post(URL, json=payload, headers=headers) as resp:
            data = await resp.json()
            return data["choices"][0]["message"]

async def chat_with_tools(user_id, text):
    history = sessions.setdefault(user_id, [])
    history.append({"role": "user", "content": text})
    del history[:-10]
    messages = [SYSTEM] + history

    msg = await ask_model(messages)
    steps = 0
    while msg.get("tool_calls") and steps < 3:
        messages.append(msg)
        for call in msg["tool_calls"]:
            name = call["function"]["name"]
            args = json.loads(call["function"]["arguments"])
            result = await TOOL_FUNCS[name](**args)
            messages.append({"role": "tool",
                             "tool_call_id": call["id"],
                             "content": result})
        msg = await ask_model(messages)
        steps += 1

    answer = msg["content"]
    history.append({"role": "assistant", "content": answer})
    return answer

@dp.message(Command("start"))
async def cmd_start(message: types.Message):
    await message.answer(
        "Бот с function calling. Модель сама решает, "
        "вызвать погоду, курс валют или просто ответить.\n"
        "/clear — очистить историю\n/help — помощь")

@dp.message(Command("help"))
async def cmd_help(message: types.Message):
    await message.answer(
        "Спросите: «погода в Минске», «курс евро», "
        "или задайте любой вопрос. Модель сама выберет, "
        "нужны ли внешние данные.")

@dp.message(Command("clear"))
async def cmd_clear(message: types.Message):
    sessions.pop(message.from_user.id, None)
    await message.answer("История очищена.")

@dp.message(F.text)
async def handle_text(message: types.Message):
    try:
        answer = await chat_with_tools(
            message.from_user.id, message.text)
    except Exception as e:
        answer = f"Ошибка: {e}"
    await message.answer(answer)

async def main():
    print("Бот запущен")
    await dp.start_polling(bot)

if __name__ == "__main__":
    asyncio.run(main())
